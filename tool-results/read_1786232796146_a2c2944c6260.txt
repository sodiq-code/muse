   433→function voiceBarColor(value: number): string {
   434→  if (value >= 70) return 'bg-emerald-500';
   435→  if (value >= 40) return 'bg-amber-500';
   436→  return 'bg-rose-400';
   437→}

   439→function voiceBadgeColor(value: number): string {
   440→  if (value >= 70) return 'bg-emerald-500/10 text-emerald-400 border-emerald-500/20';
   441→  if (value >= 40) return 'bg-amber-500/10 text-amber-400 border-amber-500/20';
   442→  return 'bg-rose-500/10 text-rose-400 border-rose-500/20';
   443→}
   444→
   445→function statusBadgeStyle(status: string): string {
   446→  switch (status) {
   447→    case 'published': return 'bg-emerald-500/10 text-emerald-400 border-emerald-500/20';
   448→    case 'drafting': return 'bg-amber-500/10 text-amber-400 border-amber-500/20';
   449→    case 'idea': return 'bg-violet-500/10 text-violet-400 border-violet-500/20';
   450→    case 'archived': return 'bg-slate-500/10 text-slate-400 border-slate-500/20';
   451→    default: return '';
   452→  }
   453→}
   454→
   455→function confidenceBadgeStyle(confidence: string): string {
   456→  switch (confidence) {
   457→    case 'high': return 'bg-emerald-500/10 text-emerald-400 border-emerald-500/20';
   458→    case 'medium': return 'bg-amber-500/10 text-amber-400 border-amber-500/20';
   459→    case 'low': return 'bg-rose-500/10 text-rose-400 border-rose-500/20';
   460→    default: return '';
   461→  }
