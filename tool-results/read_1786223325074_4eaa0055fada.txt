   146→function truncateId(id: string, len = 8): string {

   150→function statusIcon(status: 'pass' | 'fail' | 'warn') {
   151→  if (status === 'pass') return <CheckCircle2 className="size-4 text-emerald-500" />;
   152→  if (status === 'fail') return <AlertTriangle className="size-4 text-red-500" />;
   153→  return <CircleDashed className="size-4 text-amber-500" />;
   154→}
