   323→function voiceBarColor(value: number): string {
   324→  if (value >= 70) return 'bg-emerald-500';
   325→  if (value >= 40) return 'bg-amber-500';
   326→  return 'bg-rose-400';
   327→}
   328→
   329→function voiceBadgeColor(value: number): string {
   330→  if (value >= 70) return 'bg-emerald-500/10 text-emerald-400 border-emerald-500/20';
   331→  if (value >= 40) return 'bg-amber-500/10 text-amber-400 border-amber-500/20';
   332→  return 'bg-rose-500/10 text-rose-400 border-rose-500/20';
   333→}
   334→
   335→function statusBadgeStyle(status: string): string {
   336→  switch (status) {
   337→    case 'published': return 'bg-emerald-500/10 text-emerald-400 border-emerald-500/20';
   338→    case 'drafting': return 'bg-amber-500/10 text-amber-400 border-amber-500/20';
   339→    case 'idea': return 'bg-violet-500/10 text-violet-400 border-violet-500/20';
   340→    case 'archived': return 'bg-slate-500/10 text-slate-400 border-slate-500/20';
   341→    default: return '';
   342→  }
   343→}
   344→
   345→function confidenceBadgeStyle(confidence: string): string {
   346→  switch (confidence) {
   347→    case 'high': return 'bg-emerald-500/10 text-emerald-400 border-emerald-500/20';
   348→    case 'medium': return 'bg-amber-500/10 text-amber-400 border-amber-500/20';
   349→    case 'low': return 'bg-rose-500/10 text-rose-400 border-rose-500/20';
   350→    default: return '';
   351→  }
   352→}
