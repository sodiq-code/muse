    68→function splitSentences(text: string): string[] {
    69→  // Split on sentence-ending punctuation, keeping meaningful sentences
    70→  return text
    71→    .split(/[.!?]+/)
    72→    .map(s => s.trim())
    73→    .filter(s => s.length > 5);
    74→}
    75→
    76→function countWords(text: string): number {
    77→  return text.split(/\s+/).filter(w => w.length > 0).length;
    78→}
    79→
    80→function countSyllables(word: string): number {
    81→  // Simple syllable estimation
    82→  const lower = word.toLowerCase();
    83→  if (lower.length <= 3) return 1;
    84→  let count = lower.replace(/(?:[^laeiouy]es|ed|[^laeiouy]e)$/, '').match(/[aeiouy]{1,2}/g)?.length ?? 1;
    85→  return Math.max(1, count);
    86→}

    92→/**
    93→ * DIRECTNESS (0-100): How direct/blunt vs hedging/qualifying
    94→ * High: Short declarative sentences, imperative mood, "here's why", "stop doing X"
    95→ * Low: "maybe", "perhaps", "I think", "it seems", qualifiers, hedging
    96→ */
    97→function analyzeDirectness(text: string, sentences: string[]): VoiceDimensionDetail {
    98→  const indicators: string[] = [];
    99→  let score = 50; // baseline
   100→
   101→  const lower = text.toLowerCase();
   102→
   103→  // Direct language markers
   104→  if (/\b(here'?s?\s+why|here'?s?\s+how|here'?s?\s+the)\b/i.test(text)) { score += 8; indicators.push("'here's why/how' pattern"); }
   105→  if (/\bstop\s+(doing|using|believing|writing)/i.test(text)) { score += 10; indicators.push("imperative 'stop'"); }
   106→  if (/\bdon'?t\b/i.test(text)) { score += 3; indicators.push("'don't' directive"); }
   107→  if (/\bmust\b|\bneed\s+to\b|\bhave\s+to\b|\bshould\b/i.test(text)) { score += 5; indicators.push("obligation modal"); }
   108→  if (/\bwrong\b/i.test(text)) { score += 5; indicators.push("'wrong' assertion"); }
   109→  if (/\bnever\b|\balways\b|\bevery\b/i.test(text)) { score += 4; indicators.push("absolute language"); }
   110→
   111→  // Hedging language markers (reduce directness)
   112→  const hedgeCount = (lower.match(/\b(maybe|perhaps|might|could|possibly|seems?|appears?|probably|likely|sort\s+of|kind\s+of|fairly|quite|rather|somewhat)\b/g) ?? []).length;
   113→  if (hedgeCount > 0) { score -= hedgeCount * 3; indicators.push(`${hedgeCount} hedging words`); }
   114→
   115→  if (/\bi\s+think\b|\bi\s+believe\b|\bi\s+feel\s+like\b/i.test(text)) { score -= 5; indicators.push("'I think/believe' hedging"); }
   116→
   117→  // Short sentences = more direct
   118→  const avgSentenceLen = sentences.length > 0
   119→    ? sentences.reduce((s, sent) => s + countWords(sent), 0) / sentences.length
   120→    : 15;
   121→  if (avgSentenceLen < 10) { score += 8; indicators.push('short sentences (<10 words avg)'); }
   122→  else if (avgSentenceLen < 15) { score += 4; indicators.push('moderate sentences (10-15 words avg)'); }
   123→  else if (avgSentenceLen > 25) { score -= 5; indicators.push('long sentences (>25 words avg)'); }
   124→
   125→  // Imperative mood at sentence start
   126→  const imperativeCount = sentences.filter(s => /^(go|stop|start|try|use|build|create|don'?t|never|always|avoid|skip|check|read|watch|learn)\b/i.test(s)).length;
   127→  if (imperativeCount > 0) { score += imperativeCount * 4; indicators.push(`${imperativeCount} imperative sentences`); }
   128→
   129→  return {
   130→    name: 'Directness',
   131→    key: 'directness',
   132→    score: Math.max(0, Math.min(100, score)),
   133→    indicators,
   134→    reasoning: score > 70 ? 'Highly direct — uses imperatives, assertions, and minimal hedging'
   135→      : score > 40 ? 'Moderately direct — some hedging but generally assertive'
   136→        : 'Low directness — heavy use of qualifiers and hedging language',
   137→  };
   138→}
   139→
   140→/**
   141→ * TECHNICAL DEPTH (0-100): Technical jargon/sophistication vs casual/simple
   142→ * High: Code terms, API names, architecture patterns, specific tool names
   143→ * Low: Simple vocabulary, general terms, no code references
   144→ */
   145→function analyzeTechnicalDepth(text: string, _sentences: string[]): VoiceDimensionDetail {
   146→  const indicators: string[] = [];
   147→  let score = 30; // baseline (most content is somewhat technical)
   148→
   149→  // Technical terms
   150→  const techTerms = (text.match(/\b(api|sdk|cli|git|docker|kubernetes|prisma|nextjs|react|typescript|javascript|node|webpack|vite|rust|golang|python|sql|nosql|rest|graphql|websocket|oauth|jwt|tls|ssl|dns|tcp|udp|http|grpc|protobuf|json|yaml|toml|csv|regex|async|await|promise|callback|middleware|decorator|factory|singleton|observer|middleware|repository|controller|service|component|hook|render|state|props|context|redux|zustand|prisma|schema|migration|seed|crud|orm|dao|dto|vo|entity|model|view)\b/gi) ?? []).length;
   151→  if (techTerms > 0) { score += Math.min(30, techTerms * 3); indicators.push(`${techTerms} technical terms`); }
   152→
   153→  // Code-like patterns (backticks, brackets, camelCase)
   154→  const codePatterns = (text.match(/[`{}[\]()]/g) ?? []).length;
   155→  if (codePatterns > 5) { score += 8; indicators.push('code syntax patterns'); }
   156→
   157→  const camelCase = (text.match(/\b[a-z][a-zA-Z0-9]*[A-Z][a-zA-Z0-9]*\b/g) ?? []).length;
   158→  if (camelCase > 0) { score += Math.min(10, camelCase * 2); indicators.push(`${camelCase} camelCase terms`); }
   159→
   160→  // Architecture/pattern language
   161→  if (/\b(architecture|pattern|paradigm|abstraction|encapsulation|polymorphism|composition|delegation|orchestrat)\b/i.test(text)) {
   162→    score += 10; indicators.push('architecture/pattern language');
   163→  }
   164→
   165→  // Specific numbers with technical context (version numbers, ports, etc.)
   166→  if (/\bv?\d+\.\d+\.\d+\b/.test(text)) { score += 5; indicators.push('version numbers'); }
   167→  if (/\bport\s+\d+\b|\b:\d{4}\b/.test(text)) { score += 3; indicators.push('port references'); }
   168→
   169→  // Casual language (reduces technical depth)
   170→  const casualCount = (text.match(/\b(awesome|cool|nice|great|amazing|super|pretty|basically|just|stuff|thing|thingy|whatchamacallit)\b/gi) ?? []).length;
   171→  if (casualCount > 3) { score -= 5; indicators.push(`${casualCount} casual words`); }
   172→
   173→  return {
   174→    name: 'Technical Depth',
   175→    key: 'technicalDepth',
   176→    score: Math.max(0, Math.min(100, score)),
   177→    indicators,
   178→    reasoning: score > 70 ? 'Deeply technical — uses specific tools, patterns, and code references'
   179→      : score > 40 ? 'Moderately technical — some jargon but accessible'
   180→        : 'Low technical depth — casual/simple vocabulary',
   181→  };
   182→}
   183→
   184→/**
   185→ * HUMOR (0-100): Humor/playfulness vs serious
   186→ * High: Jokes, self-deprecation, witty phrasing, emojis in context
   187→ * Low: Serious, formal, no levity
   188→ */
   189→function analyzeHumor(text: string, _sentences: string[]): VoiceDimensionDetail {
   190→  const indicators: string[] = [];
   191→  let score = 15; // baseline (most tech content is somewhat serious)
   192→
   193→  const lower = text.toLowerCase();
   194→
   195→  // Self-deprecation
   196→  if (/\bi\s+(failed|broke|messed\s+up|screwed|ruined|couldn'?t|struggled|hated|wanted\s+to\s+(quit|give\s+up))\b/i.test(text)) {
   197→    score += 15; indicators.push('self-deprecating humor');
   198→  }
   199→
   200→  // Witty/sarcastic patterns
   201→  if (/\b(spoiler|plot\s+twist|fun\s+fact|pro\s+tip|protip|spoiler\s+alert)\b/i.test(text)) {
   202→    score += 10; indicators.push('witty/sarcastic pattern');
   203→  }
   204→
   205→  // Exaggeration for comedic effect
   206→  if (/\b(throw\s+your\s+(keyboard|monitor|computer|laptop)|want\s+to\s+scream|pull\s+your\s+hair|rage)\b/i.test(text)) {
   207→    score += 12; indicators.push('comedic exaggeration');
   208→  }
   209→
   210→  // "Of course" / "Naturally" used sarcastically
   211→  if (/\b(of\s+course|naturally|obviously|clearly)\b.*\b(but|except|unless|wrong|not)\b/i.test(text)) {
   212→    score += 8; indicators.push('sarcastic setup');
   213→  }
   214→
   215→  // Colloquial humor
   216→  if (/\b(yeah|yep|nope|uh|um|oh|wow|whoa|ha|lol|smh|facepalm)\b/i.test(text)) {
   217→    score += 5; indicators.push('colloquial/humorous interjection');
   218→  }
   219→
   220→  // Metaphorical humor
   221→  if (/\b(like\s+a\s+|it'?s?\s+like|imagine\s+if)\b/i.test(text) && /\b(intern|junior|toddler|chaos|disaster|nightmare|circus)\b/i.test(text)) {
   222→    score += 8; indicators.push('humorous metaphor');
   223→  }
   224→
   225→  // Parenthetical asides
   226→  const parenCount = (text.match(/\([^)]{10,}\)/g) ?? []).length;
   227→  if (parenCount > 0) { score += parenCount * 3; indicators.push(`${parenCount} parenthetical asides`); }
   228→
   229→  return {
   230→    name: 'Humor',
   231→    key: 'humor',
   232→    score: Math.max(0, Math.min(100, score)),
   233→    indicators,
   234→    reasoning: score > 50 ? 'Humorous — uses wit, self-deprecation, and playful language'
   235→      : score > 25 ? 'Some humor — occasional light moments but mostly serious'
   236→        : 'Serious — minimal humor, formal tone',
   237→  };
   238→}
   239→
   240→/**
   241→ * HYPE (0-100): Hype/exaggeration vs measured/grounded
   242→ * High: "revolutionary", "game-changing", "mind-blowing", ALL CAPS, excessive !!!
   243→ * Low: Measured claims, specific numbers, caveats, "it depends"
   244→ */
   245→function analyzeHype(text: string, _sentences: string[]): VoiceDimensionDetail {
   246→  const indicators: string[] = [];
   247→  let score = 10; // baseline (default is low hype)
   248→
   249→  // Hype words
   250→  const hypeWords = (text.match(/\b(revolutionary|game[- ]changing|mind[- ]blowing|insane|insane|unbelievable|incredible|earth[- ]shattering|groundbreaking|world[- ]class|disruptive|paradigm[- ]shift|transformative|life[- ]changing|jaw[- ]dropping)\b/gi) ?? []).length;
   251→  if (hypeWords > 0) { score += hypeWords * 12; indicators.push(`${hypeWords} hype words`); }
   252→
   253→  // "X will change everything" / "the future of X"
   254→  if (/\bwill\s+change\s+everything\b|\bfuture\s+of\b/i.test(text)) { score += 15; indicators.push("'change everything' / 'future of' pattern"); }
   255→
   256→  // Exclamation marks
   257→  const exclCount = (text.match(/!/g) ?? []).length;
   258→  if (exclCount > 2) { score += Math.min(15, exclCount * 3); indicators.push(`${exclCount} exclamation marks`); }
   259→
   260→  // ALL CAPS words (3+ letters)
   261→  const capsWords = (text.match(/\b[A-Z]{3,}\b/g) ?? []).filter(w => !['API', 'SDK', 'CLI', 'HTTP', 'REST', 'SQL', 'CSS', 'HTML', 'JSON', 'YAML', 'TCP', 'UDP', 'DNS', 'TLS', 'SSL', 'JWT', 'ORM', 'DAO', 'DTO', 'CRUD'].includes(w));
   262→  if (capsWords.length > 2) { score += capsWords.length * 3; indicators.push(`${capsWords.length} ALL CAPS emphasis words`); }
   263→
   264→  // "You NEED" / "Must have" / urgency language
   265→  if (/\byou\s+(need|must|have\s+to)\b|\burgently\b|\bnow\b|\bright\s+now\b/i.test(text)) {
   266→    score += 8; indicators.push('urgency language');
   267→  }
   268→
   269→  // Measured language (reduces hype)
   270→  if (/\b(it\s+depends|caveat|limitation|trade[- ]off|downside|important\s+caveat|with\s+reservations)\b/i.test(text)) {
   271→    score -= 10; indicators.push('measured/caveat language');
   272→  }
   273→  if (/\b(specifically|exactly|precisely|in\s+practice|in\s+reality|realistically)\b/i.test(text)) {
   274→    score -= 5; indicators.push('grounding language');
   275→  }
   276→
   277→  return {
   278→    name: 'Hype',
   279→    key: 'hype',
   280→    score: Math.max(0, Math.min(100, score)),
   281→    indicators,
   282→    reasoning: score > 50 ? 'High hype — uses superlatives, urgency, and exaggerated claims'
   283→      : score > 25 ? 'Some hype — occasional enthusiasm but generally grounded'
   284→        : 'Low hype — measured, grounded claims with specifics',
   285→  };
   286→}
   287→
   288→/**
   289→ * STORYTELLING (0-100): Narrative/story-driven vs factual/list
   290→ * High: "I", personal anecdotes, narrative arc, "then", temporal markers
   291→ * Low: Bullet points, numbered lists, definitions, pure facts
   292→ */
   293→function analyzeStorytelling(text: string, sentences: string[]): VoiceDimensionDetail {
   294→  const indicators: string[] = [];
   295→  let score = 30; // baseline
   296→
   297→  // First person narrative
   298→  const iCount = (text.match(/\bI\s+/g) ?? []).length;
   299→  if (iCount > 3) { score += Math.min(20, iCount * 3); indicators.push(`${iCount} first-person references`); }
   300→
   301→  // Temporal/narrative markers
   302→  const narrativeMarkers = (text.match(/\b(last\s+(week|month|year|night|time)|yesterday|earlier|then|after\s+that|next|finally|eventually|at\s+first|in\s+the\s+end|once|one\s+day)\b/gi) ?? []).length;
   303→  if (narrativeMarkers > 0) { score += Math.min(15, narrativeMarkers * 4); indicators.push(`${narrativeMarkers} narrative markers`); }
   304→
   305→  // Problem → resolution arc
   306→  if (/\b(problem|issue|bug|error|fail|broke|struggl)\b/i.test(text) && /\b(fix|solv|resolv|work|success|figur|discover)\b/i.test(text)) {
   307→    score += 12; indicators.push('problem → resolution arc');
   308→  }
   309→
   310→  // "Let me tell you" / "Story time"
   311→  if (/\b(let\s+me\s+tell|story\s+time|here'?s?\s+what\s+happened|so\s+here'?s?\s+the\s+story)\b/i.test(text)) {
   312→    score += 10; indicators.push('storytelling setup');
   313→  }
   314→
   315→  // Dialogue/quotes
   316→  if (/"[^"]{10,}"/.test(text) || /'[^']{10,}'/.test(text)) {
   317→    score += 5; indicators.push('quoted dialogue');
   318→  }
   319→
   320→  // Anti-storytelling: numbered lists, bullet points reduce storytelling score
   321→  const listItems = (text.match(/^\s*[-*•]\s|\n\s*[-*•]\s|\n\s*\d+[.)]\s/gm) ?? []).length;
   322→  if (listItems > 3) { score -= 8; indicators.push(`${listItems} list items (less narrative)`); }
   323→
   324→  // "Step 1, Step 2" pattern reduces storytelling
   325→  if (/step\s+\d/i.test(text)) { score -= 5; indicators.push('step-by-step (less narrative)'); }
   326→
   327→  return {
   328→    name: 'Storytelling',
   329→    key: 'storytelling',
   330→    score: Math.max(0, Math.min(100, score)),
   331→    indicators,
   332→    reasoning: score > 60 ? 'Strong storyteller — uses personal anecdotes and narrative arcs'
   333→      : score > 35 ? 'Some storytelling — mixes narrative with factual content'
   334→        : 'Low storytelling — primarily factual/list-based content',
   335→  };
   336→}
   337→
   338→/**
   339→ * SENTENCE LENGTH (0-100): Long complex vs short punchy sentences
   340→ * High: Average sentence > 20 words, complex clauses
   341→ * Low: Short punchy sentences, fragments
   342→ */
   343→function analyzeSentenceLength(text: string, sentences: string[]): VoiceDimensionDetail {
   344→  const indicators: string[] = [];
   345→
   346→  if (sentences.length === 0) {
   347→    return {
   348→      name: 'Sentence Length',
   349→      key: 'sentenceLength',
   350→      score: 50,
   351→      indicators: ['no sentences detected'],
   352→      reasoning: 'No sentences detected for analysis',
   353→    };
   354→  }
   355→
   356→  const wordCounts = sentences.map(s => countWords(s));
   357→  const avgLen = wordCounts.reduce((s, l) => s + l, 0) / wordCounts.length;
   358→  const maxLen = Math.max(...wordCounts);
   359→  const shortPct = wordCounts.filter(l => l < 8).length / wordCounts.length;
   360→
   361→  // Map average sentence length to 0-100 score
   362→  // 5 words = ~10, 15 words = ~50, 30 words = ~90
   363→  let score = Math.min(100, Math.max(0, (avgLen - 3) * 4));
   364→
   365→  if (avgLen > 20) indicators.push(`long sentences (avg ${avgLen.toFixed(1)} words)`);
   366→  else if (avgLen > 12) indicators.push(`moderate sentences (avg ${avgLen.toFixed(1)} words)`);
   367→  else indicators.push(`short sentences (avg ${avgLen.toFixed(1)} words)`);
   368→
   369→  if (maxLen > 30) { score += 5; indicators.push(`max sentence ${maxLen} words`); }
   370→  if (shortPct > 0.3) { score -= 8; indicators.push(`${(shortPct * 100).toFixed(0)}% short punchy sentences`); }
   371→
   372→  // Subordinate clauses increase complexity
   373→  const clauseMarkers = (text.match(/\b(although|because|since|while|whereas|unless|until|whenever|wherever|whether|which|that|who)\b/gi) ?? []).length;
   374→  if (clauseMarkers > 2) { score += 5; indicators.push(`${clauseMarkers} subordinate clauses`); }
   375→
   376→  return {
   377→    name: 'Sentence Length',
   378→    key: 'sentenceLength',
   379→    score: Math.max(0, Math.min(100, Math.round(score))),
   380→    indicators,
   381→    reasoning: score > 60 ? 'Long, complex sentences — elaborate and detailed'
   382→      : score > 35 ? 'Moderate sentence length — balanced readability'
   383→        : 'Short, punchy sentences — concise and direct',
   384→  };
   385→}
   386→
   387→/**
   388→ * CTA INTENSITY (0-100): Strong CTAs vs subtle/absent
   389→ * High: "Subscribe now", "Follow me", "Click here", multiple CTAs
   390→ * Low: No CTAs, soft suggestions, "if you want"
   391→ */
   392→function analyzeCtaIntensity(text: string, _sentences: string[]): VoiceDimensionDetail {
   393→  const indicators: string[] = [];
   394→  let score = 10; // baseline (most content has minimal CTAs)
   395→
   396→  // Strong CTA patterns
   397→  if (/\b(subscribe\s+now|follow\s+me|click\s+(here|below)|sign\s+up|join\s+(us|me|the)|download\s+(now|free|today))\b/i.test(text)) {
   398→    score += 20; indicators.push('strong direct CTA');
   399→  }
   400→
   401→  // Moderate CTAs
   402→  if (/\b(hit\s+)?subscribe\b|\bfollow\b|\blike\s+(and\s+subscribe|this\s+video)?\b/i.test(text)) {
   403→    score += 10; indicators.push('standard subscribe/follow CTA');
   404→  }
   405→
   406→  // "If you liked this" / soft CTA
   407→  if (/\bif\s+you\s+(liked|enjoyed|found\s+this)\b/i.test(text)) {
   408→    score += 8; indicators.push('soft conditional CTA');
   409→  }
   410→
   411→  // "Let me know" / engagement CTA
   412→  if (/\b(let\s+me\s+know|drop\s+a\s+comment|comment\s+below|share\s+your|tell\s+me)\b/i.test(text)) {
   413→    score += 8; indicators.push('engagement CTA');
   414→  }
   415→
   416→  // Multiple CTAs
   417→  const ctaPatterns = [
   418→    /\bsubscribe\b/i,
   419→    /\bfollow\b/i,
   420→    /\blike\b/i,
   421→    /\bcomment\b/i,
   422→    /\bshare\b/i,
   423→    /\bclick\b/i,
   424→    /\bsign\s+up\b/i,
   425→    /\bjoin\b/i,
   426→    /\bdownload\b/i,
   427→    /\blink\s+(in|below|below)\b/i,
   428→  ];
   429→  const ctaCount = ctaPatterns.filter(p => p.test(text)).length;
   430→  if (ctaCount >= 3) { score += 10; indicators.push(`${ctaCount} different CTA types`); }
   431→
   432→  // Urgency modifiers on CTAs
   433→  if (/\b(now|today|right\s+now|don'?t\s+wait|hurry)\b/i.test(text) && ctaCount > 0) {
   434→    score += 8; indicators.push('urgency modifier on CTA');
   435→  }
   436→
   437→  // "No pressure" / anti-CTA
   438→  if (/\b(no\s+pressure|only\s+if\s+you\s+want|optional|up\s+to\s+you)\b/i.test(text)) {
   439→    score -= 5; indicators.push('soft/optional framing');
   440→  }
   441→
   442→  return {
   443→    name: 'CTA Intensity',
   444→    key: 'ctaIntensity',
   445→    score: Math.max(0, Math.min(100, score)),
   446→    indicators,
   447→    reasoning: score > 50 ? 'Strong CTAs — multiple direct calls to action'
   448→      : score > 25 ? 'Moderate CTAs — some calls to action, not aggressive'
   449→        : 'Low CTA intensity — minimal or soft calls to action',
   450→  };
   451→}

   453→// ---------------------------------------------------------------------------
   454→// Main Analysis Function
   455→// ---------------------------------------------------------------------------
   456→
   457→export function analyzeVoice(text: string): VoiceAnalysisResult {
   458→  const sentences = splitSentences(text);
   459→  const wordCount = countWords(text);
   460→
   461→  const dimensions: VoiceDimensionDetail[] = [
   462→    analyzeDirectness(text, sentences),
   463→    analyzeTechnicalDepth(text, sentences),
   464→    analyzeHumor(text, sentences),
   465→    analyzeHype(text, sentences),
   466→    analyzeStorytelling(text, sentences),
   467→    analyzeSentenceLength(text, sentences),
   468→    analyzeCtaIntensity(text, sentences),
   469→  ];
   470→
   471→  const profile: VoiceProfile = {
   472→    directness: dimensions.find(d => d.key === 'directness')!.score,
   473→    technicalDepth: dimensions.find(d => d.key === 'technicalDepth')!.score,
   474→    humor: dimensions.find(d => d.key === 'humor')!.score,
   475→    hype: dimensions.find(d => d.key === 'hype')!.score,
   476→    storytelling: dimensions.find(d => d.key === 'storytelling')!.score,
   477→    sentenceLength: dimensions.find(d => d.key === 'sentenceLength')!.score,
   478→    ctaIntensity: dimensions.find(d => d.key === 'ctaIntensity')!.score,
   479→  };
   480→
   481→  return {
   482→    profile,
   483→    sampleText: text.slice(0, 200) + (text.length > 200 ? '...' : ''),
   484→    wordCount,
   485→    sentenceCount: sentences.length,
   486→    dimensions,
   487→  };
   488→}

   490→// ---------------------------------------------------------------------------
   491→// Voice Match — compare content voice to stored voice profile
   492→// ---------------------------------------------------------------------------
   493→
   494→export function computeVoiceMatch(content: VoiceProfile, target: VoiceProfile): VoiceMatchResult {
   495→  const keys: (keyof VoiceProfile)[] = [
   496→    'directness', 'technicalDepth', 'humor', 'hype',
   497→    'storytelling', 'sentenceLength', 'ctaIntensity',
   498→  ];
   499→
   500→  // Weight dimensions by importance for voice matching
   501→  // Directness and technicalDepth are most important for identity
   502→  const weights: Record<keyof VoiceProfile, number> = {
   503→    directness: 0.25,
   504→    technicalDepth: 0.20,
   505→    humor: 0.10,
   506→    hype: 0.15,
   507→    storytelling: 0.12,
   508→    sentenceLength: 0.08,
   509→    ctaIntensity: 0.10,
   510→  };
   511→
   512→  const dimensionScores: Record<keyof VoiceProfile, number> = {} as Record<keyof VoiceProfile, number>;
   513→  const mismatches: string[] = [];
   514→
   515→  for (const key of keys) {
   516→    // Similarity: 1 - |a - b| / 100
   517→    const similarity = 1 - Math.abs(content[key] - target[key]) / 100;
   518→    dimensionScores[key] = Math.round(similarity * 100) / 100;
   519→
   520→    // Flag significant mismatches (>30 points off)
   521→    if (Math.abs(content[key] - target[key]) > 30) {
   522→      const direction = content[key] > target[key] ? 'higher' : 'lower';
   523→      mismatches.push(`${key} is ${direction} than profile (${content[key]} vs ${target[key]})`);
   524→    }
   525→  }
   526→
   527→  // Weighted overall score
   528→  let overallScore = 0;
   529→  let totalWeight = 0;
   530→  for (const key of keys) {
   531→    overallScore += dimensionScores[key] * weights[key];
   532→    totalWeight += weights[key];
   533→  }
   534→  overallScore = overallScore / totalWeight;
   535→
   536→  // Generate summary
   537→  const matchPct = (overallScore * 100).toFixed(1);
   538→  let matchSummary = `Voice match: ${matchPct}%`;
   539→  if (mismatches.length > 0) {
   540→    matchSummary += `. Mismatches: ${mismatches.join('; ')}`;
   541→  } else {
   542→    matchSummary += '. All dimensions within target range.';
   543→  }
   544→
   545→  return {
   546→    score: Math.round(overallScore * 100) / 100,
   547→    dimensionScores,
   548→    mismatches,
   549→    matchSummary,
   550→  };
   551→}

   557→/** Get the voice profile for a creator */
   558→export async function getVoiceProfile(creatorId: string): Promise<VoiceProfile | null> {
   559→  const creator = await db.creator.findUnique({
   560→    where: { id: creatorId },
   561→    select: { voiceProfile: true },
   562→  });
   563→  if (!creator?.voiceProfile) return null;
   564→  return JSON.parse(creator.voiceProfile);
   565→}
   566→
   567→/** Update voice profile — merges new analysis with existing profile using weighted average */
   568→export async function updateVoiceProfile(
   569→  creatorId: string,
   570→  newProfile: VoiceProfile,
   571→  weight: number = 0.3 // Weight for new data (0-1); existing gets (1-weight)
   572→): Promise<VoiceProfile> {
   573→  const existing = await getVoiceProfile(creatorId);
   574→
   575→  let merged: VoiceProfile;
   576→  if (existing) {
   577→    // Weighted merge: existing * (1-weight) + new * weight
   578→    const keys: (keyof VoiceProfile)[] = [
   579→      'directness', 'technicalDepth', 'humor', 'hype',
   580→      'storytelling', 'sentenceLength', 'ctaIntensity',
   581→    ];
   582→    merged = { ...existing };
   583→    for (const key of keys) {
   584→      merged[key] = Math.round(existing[key] * (1 - weight) + newProfile[key] * weight);
   585→    }
   586→  } else {
   587→    merged = newProfile;
   588→  }
   589→
   590→  // Update in database
   591→  await db.creator.update({
   592→    where: { id: creatorId },
   593→    data: { voiceProfile: JSON.stringify(merged) },
   594→  });
   595→
   596→  // Log memory events for changed dimensions
   597→  if (existing) {
   598→    const keys: (keyof VoiceProfile)[] = [
   599→      'directness', 'technicalDepth', 'humor', 'hype',
   600→      'storytelling', 'sentenceLength', 'ctaIntensity',
   601→    ];
   602→    for (const key of keys) {
   603→      if (Math.abs(merged[key] - existing[key]) >= 2) {
   604→        await logMemoryEvent({
   605→          creatorId,
   606→          category: 'identity',
   607→          key: `voice_${key}`,
   608→          value: String(merged[key]),
   609→          source: 'analytics',
   610→          confidence: 0.7, // Voice analysis is inferred, not directly stated
   611→        });
   612→      }
   613→    }
   614→  }
   615→
   616→  // Audit event
   617→  await db.auditEvent.create({
   618→    data: {
   619→      creatorId,
   620→      actor: 'system',
   621→      action: 'update',
   622→      targetType: 'voice_profile',
   623→      targetId: creatorId,
   624→      delta: JSON.stringify({
   625→        before: existing,
   626→        after: merged,
   627→        weight,
   628→      }),
   629→    },
   630→  });
   631→
   632→  return merged;
   633→}
   634→
   635→/** Analyze text and update voice profile in one step */
   636→export async function analyzeAndUpdateVoice(
   637→  creatorId: string,
   638→  text: string,
   639→  mergeWeight: number = 0.3
   640→): Promise<{ analysis: VoiceAnalysisResult; updatedProfile: VoiceProfile; match?: VoiceMatchResult }> {
   641→  const analysis = analyzeVoice(text);
   642→  const existing = await getVoiceProfile(creatorId);
   643→
   644→  const updatedProfile = await updateVoiceProfile(creatorId, analysis.profile, mergeWeight);
   645→
   646→  let match: VoiceMatchResult | undefined;
   647→  if (existing) {
   648→    match = computeVoiceMatch(analysis.profile, existing);
   649→  }
   650→
   651→  return { analysis, updatedProfile, match };
   652→}
