   171→  creatorId: string

   173→  const startedAt = new Date();
   174→  const cycleStart = Date.now();

   176→  // ── Create AutonomousRun record ──────────────────────────────────────────
   177→  const autonomousRun = await db.autonomousRun.create({
   178→    data: {
   179→      trigger: 'scheduled',

   194→    const stepStart = Date.now();
   195→    const auditEventId = await createAuditEvent(
   196→      creatorId,
   197→      'muse',
   198→      'overnight_wake',
   199→      'autonomous_run',
   200→      autonomousRun.id,
   201→      { message: 'Muse woke up for overnight cycle (Passive Autonomous Soul)', trigger: 'scheduled' }

   205→      step: 1,
   206→      name: 'Wake',
   207→      status: 'complete',
   208→      duration,

   210→      data: { message: 'Muse woke up for overnight cycle' },
   211→    });
   212→  } catch (error) {
   213→    cycleSuccess = false;
   214→    const stepStart = Date.now();

   216→      creatorId,
   217→      'muse',
   218→      'overnight_wake_failed',
   219→      'autonomous_run',
   220→      autonomousRun.id,
   221→      { error: String(error) }
   222→    );
   223→    steps.push({
   224→      step: 1,
   225→      name: 'Wake',
   226→      status: 'failed',
   227→      duration: Date.now() - stepStart,

   231→  }

   245→      'overnight_review_signals',

   255→        isHonest: learningResult.honestyReport.isHonest,
   256→        insights,
   257→      }
   258→    );
   259→
   260→    const duration = Date.now() - stepStart;
   261→    steps.push({
   262→      step: 2,
   263→      name: 'Review Signals',
   264→      status: 'complete',
   265→      duration,
   266→      auditEventId,
   267→      data: {
   268→        observations: learningResult.loopResult.observations.length,
   269→        recommendations: recommendationsCount,
   270→        confidence,
   271→        dataPoints: learningResult.loopResult.totalDataPoints,

   279→    const auditEventId = await createAuditEvent(

   281→      'muse',

   285→      { error: String(error), note: 'Non-fatal: continuing cycle without learning data' }
   286→    );
   287→    steps.push({
   288→      step: 2,
   289→      name: 'Review Signals',
   290→      status: 'failed',
   291→      duration: Date.now() - stepStart,

   323→      }

   328→      step: 3,

   451→    // Add learning engine insights

   565→      'muse',

   579→
   580→    // Still complete the autonomous run

   582→    const totalDuration = Date.now() - cycleStart;
   583→
   584→    const fallbackBrief: MorningBriefResult = {

   623→      completedAt: completedAt.toISOString(),
   624→      totalDuration,
   625→    };
   626→  }
   627→}

   629→// ---------------------------------------------------------------------------
   630→// Helper: Build brief summary text
   631→// ---------------------------------------------------------------------------
   632→
   633→function buildBriefSummary(
   634→  draftTitle: string | null,
   635→  draftScore: number | null,
   636→  recommendationsCount: number,
   637→  newInsights: string[],
   638→  delegationSuccess: boolean
   639→): string {
   640→  const parts: string[] = [];
   641→
   642→  if (draftTitle) {
   643→    parts.push(`Draft ready for review: "${draftTitle}"`);
   644→    if (draftScore != null) {
   645→      parts.push(`Evaluation score: ${draftScore}/100`);
   646→    }
   647→  } else {
   648→    parts.push('No draft produced overnight');
   649→  }
   650→
   651→  if (recommendationsCount > 0) {
   652→    parts.push(`${recommendationsCount} pending recommendations`);
   653→  }
   654→
   655→  if (newInsights.length > 0) {
   656→    parts.push(`${newInsights.length} new insights from learning engine`);
   657→  }
   658→
   659→  if (delegationSuccess) {
   660→    parts.push('Delegation beat completed successfully');
   661→  } else {
   662→    parts.push('Delegation beat encountered issues');
   663→  }
   664→
   665→  return parts.join('. ') + '.';
   666→}

   687→  });
   688→
   689→  // Get most recent run
   690→  const lastRun = await db.autonomousRun.findFirst({
   691→    where: { taskType: 'overnight_cycle' },
   692→    orderBy: { startedAt: 'desc' },
   693→  });
   694→
   695→  // Get pending approvals count
   696→  const pendingApprovals = await db.approval.count({
   697→    where: {
   698→      creatorId,
   699→      status: 'pending',
   700→    },
   701→  });
   702→
   703→  // Get recent runs (last 10)
   704→  const recentRuns = await db.autonomousRun.findMany({
   705→    where: { taskType: 'overnight_cycle' },
   706→    orderBy: { startedAt: 'desc' },
   707→    take: 10,
   708→  });
   709→
   710→  return {
   711→    schedule: OVERNIGHT_SCHEDULE,
   712→    lastRun: lastRun
   713→      ? {
   714→          id: lastRun.id,
   715→          status: lastRun.status,
   716→          startedAt: lastRun.startedAt.toISOString(),
   717→          completedAt: lastRun.completedAt?.toISOString() ?? null,
   718→          taskType: lastRun.taskType,
   719→          trigger: lastRun.trigger,
   720→        }
   721→      : null,
   722→    lastRunTime: lastRun?.startedAt.toISOString() ?? null,
   723→    totalRuns,
   724→    pendingApprovals,
   725→    isRunning: runningRun !== null,
   726→    recentRuns: recentRuns.map((r) => ({
   727→      id: r.id,
   728→      status: r.status,
   729→      startedAt: r.startedAt.toISOString(),
   730→      completedAt: r.completedAt?.toISOString() ?? null,
   731→      taskType: r.taskType,
   732→    })),
   733→  };
   734→}

   736→// ---------------------------------------------------------------------------
   737→// Approve Action
   738→// ---------------------------------------------------------------------------
   739→
   740→export async function approveAction(
   741→  approvalId: string,
   742→  creatorId: string
   743→): Promise<ApprovalResult> {
   744→  // Get the approval
   745→  const approval = await db.approval.findUnique({
   746→    where: { id: approvalId },
   747→  });
   748→
   749→  if (!approval) {
   750→    throw new Error(`Approval not found: ${approvalId}`);
   751→  }
   752→
   753→  if (approval.creatorId !== creatorId) {
   754→    throw new Error('Approval does not belong to this creator');
   755→  }
   756→
   757→  if (approval.status !== 'pending') {
   758→    throw new Error(`Approval is already ${approval.status}`);
   759→  }
   760→
   761→  // Update approval status
   762→  const now = new Date();
   763→  await db.approval.update({
   764→    where: { id: approvalId },
   765→    data: {
   766→      status: 'approved',
   767→      reviewedAt: now,
   768→    },
   769→  });
   770→
   771→  // Create audit event
   772→  const auditEventId = await createAuditEvent(
   773→    creatorId,
   774→    'creator',
   775→    'approve',
   776→    'approval',
   777→    approvalId,
   778→    {
   779→      approvalId,
   780→      itemType: approval.itemType,
   781→      itemId: approval.itemId,
   782→      action: approval.action,
   783→      approvedAt: now.toISOString(),
   784→    }
   785→  );
   786→
   787→  // If itemType is 'draft', log the approval on the draft
   788→  if (approval.itemType === 'draft' && approval.itemId) {
   789→    await createAuditEvent(
   790→      creatorId,
   791→      'creator',
   792→      'approve_publish',
   793→      'draft',
   794→      approval.itemId,
   795→      {
   796→        draftId: approval.itemId,
   797→        approvalId,
   798→        action: 'publish_approved',
   799→        note: 'Draft approved for publishing',
   800→      }
   801→    );
   802→  }
   803→
   804→  return {
   805→    success: true,
   806→    approvalId,
   807→    itemType: approval.itemType,
   808→    itemId: approval.itemId,
   809→    action: approval.action,
   810→    status: 'approved',
   811→    auditEventId,
   812→  };
   813→}

   815→// ---------------------------------------------------------------------------
   816→// Reject Action
   817→// ---------------------------------------------------------------------------
   818→
   819→export async function rejectAction(
   820→  approvalId: string,
   821→  creatorId: string,
   822→  reason?: string
   823→): Promise<ApprovalResult> {
   824→  // Get the approval
   825→  const approval = await db.approval.findUnique({
   826→    where: { id: approvalId },
   827→  });
   828→
   829→  if (!approval) {
   830→    throw new Error(`Approval not found: ${approvalId}`);
   831→  }
   832→
   833→  if (approval.creatorId !== creatorId) {
   834→    throw new Error('Approval does not belong to this creator');
   835→  }
   836→
   837→  if (approval.status !== 'pending') {
   838→    throw new Error(`Approval is already ${approval.status}`);
   839→  }
   840→
   841→  // Update approval status
   842→  const now = new Date();
   843→  await db.approval.update({
   844→    where: { id: approvalId },
   845→    data: {
   846→      status: 'rejected',
   847→      reviewedAt: now,
   848→    },
   849→  });
   850→
   851→  // Create audit event
   852→  const auditEventId = await createAuditEvent(
   853→    creatorId,
   854→    'creator',
   855→    'reject',
   856→    'approval',
   857→    approvalId,
   858→    {
   859→      approvalId,
   860→      itemType: approval.itemType,
   861→      itemId: approval.itemId,
   862→      action: approval.action,
   863→      rejectedAt: now.toISOString(),
   864→      reason: reason ?? 'No reason provided',
   865→    }
   866→  );
   867→
   868→  // If itemType is 'draft', create a CreatorDecision
   869→  if (approval.itemType === 'draft' && approval.itemId) {
   870→    // Look up the draft to get its contentItemId (foreign key requires valid ContentItem)
   871→    const draft = await db.draft.findUnique({ where: { id: approval.itemId } });
   872→    const contentItemId = draft?.contentItemId ?? null;
   873→
   874→    // Only create CreatorDecision if we have a valid contentItemId
   875→    if (contentItemId) {
   876→      await db.creatorDecision.create({
   877→        data: {
   878→          creatorId,
   879→          contentItemId,
   880→          decision: 'rejected',
   881→          reason: reason ?? 'Creator rejected overnight draft',
   882→        },
   883→      });
   884→    }
   885→
   886→    await createAuditEvent(
   887→      creatorId,
   888→      'creator',
   889→      'reject_publish',
   890→      'draft',
   891→      approval.itemId,
   892→      {
   893→        draftId: approval.itemId,
   894→        approvalId,
   895→        action: 'publish_rejected',
   896→        reason: reason ?? 'No reason provided',
   897→      }
   898→    );
   899→  }
   900→
   901→  return {
   902→    success: true,
   903→    approvalId,
   904→    itemType: approval.itemType,
   905→    itemId: approval.itemId,
   906→    action: approval.action,
   907→    status: 'rejected',
   908→    auditEventId,
   909→  };
   910→}

   912→// ---------------------------------------------------------------------------
   913→// Get Approval Queue
   914→// ---------------------------------------------------------------------------
   915→
   916→export async function getApprovalQueue(
   917→  creatorId: string
   918→): Promise<ApprovalQueueItem[]> {
   919→  const approvals = await db.approval.findMany({
   920→    where: {
   921→      creatorId,
   922→      status: 'pending',
   923→    },
   924→    orderBy: { createdAt: 'desc' },
   925→  });
   926→
   927→  // Resolve titles for draft approvals
   928→  const queue: ApprovalQueueItem[] = [];
   929→  for (const approval of approvals) {
   930→    const item: ApprovalQueueItem = {
   931→      id: approval.id,
   932→      itemType: approval.itemType,
   933→      itemId: approval.itemId,
   934→      action: approval.action,
   935→      status: approval.status,
   936→      createdAt: approval.createdAt.toISOString(),
   937→    };
   938→
   939→    // If it's a draft, resolve the title
   940→    if (approval.itemType === 'draft' && approval.itemId) {
   941→      const draft = await db.draft.findUnique({
   942→        where: { id: approval.itemId },
   943→      });
   944→      if (draft) {
   945→        try {
   946→          const content = JSON.parse(draft.content);
   947→          item.title = content.title ?? `Draft v${draft.version}`;
   948→        } catch {
   949→          item.title = `Draft v${draft.version}`;
   950→        }
   951→      }
   952→    }
   953→
   954→    queue.push(item);
   955→  }
   956→
   957→  return queue;
   958→}
