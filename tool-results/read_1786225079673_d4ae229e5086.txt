   225→function truncateId(id: string, len = 8): string {

   229→function statusIcon(status: 'pass' | 'fail' | 'warn') {
   230→  if (status === 'pass') return <CheckCircle2 className="size-4 text-emerald-500" />;
   231→  if (status === 'fail') return <AlertTriangle className="size-4 text-red-500" />;
   232→  return <CircleDashed className="size-4 text-amber-500" />;
   233→}
